enable

line con 0

 transport input ssh

! =====================================================
! NAT PAT
! =====================================================

access-list 1 permit 172.16.3.224 0.0.0.31

! =====================================================
! DMVPN - SPOKE BARAHONA
! =====================================================

crypto isakmp policy 10
 encryption aes 256
 hash sha256
 authentication pre-share
 group 14
 lifetime 86400
exit


crypto isakmp key EMPRESA2#DMVPN2026! address 172.16.4.145


crypto ipsec transform-set TS-DMVPN esp-aes 256 esp-sha-hmac
 mode transport
exit


crypto ipsec profile IPSEC-PROFILE-DMVPN
 set transform-set TS-DMVPN
exit


interface Tunnel0
 description DMVPN-SPOKE-BARAHONA
 ip address 172.16.4.197 255.255.255.240
 ip mtu 1400
 ip tcp adjust-mss 1360

 ip nhrp network-id 1
 ip nhrp authentication cecompe

 ip nhrp nhs 172.16.4.193
 ip nhrp map 172.16.4.193 172.16.4.145
 ip nhrp map multicast 172.16.4.145

 ip ospf network broadcast
 ip ospf priority 0

 tunnel source e0/0
 tunnel mode gre multipoint
 tunnel protection ipsec profile IPSEC-PROFILE-DMVPN

 no shutdown
exit
